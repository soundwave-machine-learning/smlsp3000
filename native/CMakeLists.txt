cmake_minimum_required(VERSION 3.20)
project(smlsp3000_native LANGUAGES CXX)
# SML SP-3000 native production core (Track A, OWN-DEC-014). C++20, no external dependencies, no fast-math,
# no floating-point contraction (the numeric contract against the Python reference requires separately rounded
# products and sums). Options are recorded in evidence/sprint_06/build_record.json by the Python driver.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
# explicit, unambiguous optimisation level for the recorded build (CMake's default Release flags would add -O3)
if(NOT MSVC)
  set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG" CACHE STRING "" FORCE)
endif()
option(SMLSP3000_NATIVE_ARCH "Tune for the build machine (-march=native); off = generic x86-64 for portability" OFF)
add_library(smlsp3000_core STATIC src/kernels.cpp src/streaming.cpp src/sp_core.cpp src/mpc_core.cpp src/engine.cpp src/host_adapter.cpp)
target_include_directories(smlsp3000_core PUBLIC include generated)
set_target_properties(smlsp3000_core PROPERTIES POSITION_INDEPENDENT_CODE ON)   # linkable into shared plugin modules (no code change)
if(MSVC)
  # /fp:precise: no contraction into FMA, no reassociation (MSVC contracts only under /fp:fast or /fp:contract)
  target_compile_options(smlsp3000_core PUBLIC /W4 /fp:precise /O2)
else()
  target_compile_options(smlsp3000_core PUBLIC -Wall -Wextra -ffp-contract=off -fno-fast-math -fexcess-precision=standard)
  if(SMLSP3000_NATIVE_ARCH)
    target_compile_options(smlsp3000_core PUBLIC -march=native)
  endif()
endif()
add_executable(smlsp3000_native tools/smlsp3000_native.cpp)
target_link_libraries(smlsp3000_native PRIVATE smlsp3000_core)
add_executable(smlsp3000_selftest tests/selftest.cpp)
target_link_libraries(smlsp3000_selftest PRIVATE smlsp3000_core)
add_executable(smlsp3000_adapter_test tests/adapter_test.cpp)
target_link_libraries(smlsp3000_adapter_test PRIVATE smlsp3000_core)
enable_testing()
add_test(NAME native_selftest COMMAND smlsp3000_selftest)
add_test(NAME host_adapter_test COMMAND smlsp3000_adapter_test)
